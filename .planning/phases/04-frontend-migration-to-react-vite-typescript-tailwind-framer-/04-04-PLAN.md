**File 3: `frontend/src/components/manual/MapPanel.tsx`** — port `ghana_cap_dashboard.html:676-693`. Use raw Leaflet (NOT `react-leaflet`) per PATTERNS.md "MapPanel.tsx" — keeps the Phase 5 Mapbox swap local. Critical: useEffect cleanup calls `map.remove()` so React StrictMode double-mount doesn't leak (RESEARCH.md Pitfall #7 generalized to Leaflet):

```typescript
import { useEffect, useRef } from 'react'
import L from 'leaflet'
import 'leaflet-draw'
import { useTheme } from '../../hooks/useTheme'

interface MapPanelProps {
  /** Called whenever the operator clicks/draws a location. lat, lon are decimal degrees. */
  onLocationChange: (lat: number, lon: number) => void
  /** When the parent tab becomes visible; triggers map.invalidateSize() so Leaflet adjusts to its container. */
  visible: boolean
  className?: string
}

const TILE_URLS = {
  dark:  'https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png',
  light: 'https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png',
}

export function MapPanel({ onLocationChange, visible, className }: MapPanelProps) {
  const containerRef = useRef<HTMLDivElement | null>(null)
  const mapRef = useRef<L.Map | null>(null)
  const tileRef = useRef<L.TileLayer | null>(null)
  const drawnItemsRef = useRef<L.FeatureGroup | null>(null)
  const { theme } = useTheme()

  // Init once on mount.
  useEffect(() => {
    if (!containerRef.current || mapRef.current) return

    const map = L.map(containerRef.current).setView([7.9465, -1.0232], 7)
    mapRef.current = map

    const tile = L.tileLayer(TILE_URLS[theme], {
      attribution: '&copy; OpenStreetMap contributors &copy; CARTO',
    }).addTo(map)
    tileRef.current = tile

    const drawnItems = new L.FeatureGroup()
    map.addLayer(drawnItems)
    drawnItemsRef.current = drawnItems

    const drawControl = new (L.Control as any).Draw({
      draw: { polyline: false, circlemarker: false },
      edit: { featureGroup: drawnItems },
    })
    map.addControl(drawControl)

    map.on((L as any).Draw.Event.CREATED, (e: any) => {
      drawnItems.clearLayers()
      drawnItems.addLayer(e.layer)
      const ll: L.LatLng =
        typeof e.layer.getLatLng === 'function'
          ? e.layer.getLatLng()
          : e.layer.getBounds().getCenter()
      onLocationChange(ll.lat, ll.lng)
    })

    return () => {
      map.remove()
      mapRef.current = null
      tileRef.current = null
      drawnItemsRef.current = null
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps -- intentional one-shot init; theme + onLocationChange handled in their own effects
  }, [])

  // Theme-reactive tile swap. Removes the old tile, adds the new one.
  // PATTERNS.md "MapPanel.tsx" — make tile reactive to useTheme().
  useEffect(() => {
    const map = mapRef.current
    if (!map) return
    if (tileRef.current) {
      map.removeLayer(tileRef.current)
      tileRef.current = null
    }
    tileRef.current = L.tileLayer(TILE_URLS[theme], {
      attribution: '&copy; OpenStreetMap contributors &copy; CARTO',
    }).addTo(map)
  }, [theme])

  // invalidateSize when the tab becomes visible — fixes the existing tab-switch bug
  // (ghana_cap_dashboard.html:640) where Leaflet renders zero-size if container is hidden.
  useEffect(() => {
    if (visible && mapRef.current) {
      const id = window.setTimeout(() => mapRef.current?.invalidateSize(), 100)
      return () => window.clearTimeout(id)
    }
  }, [visible])

  return (
    <div
      ref={containerRef}
      className={className ?? 'w-full h-[420px] rounded-xl overflow-hidden border border-white/10'}
    />
  )
}
```

**File 4: `frontend/src/components/manual/ManualEntryForm.tsx`** — controlled form. Replaces the `document.getElementById` pattern entirely; lat/lon arrive via prop from the parent (PATTERNS.md "ManualEntry.tsx"):

```typescript
import { useState } from 'react'
import { submitManualAlert } from '../../api/alerts'
import { ApiError } from '../../api/client'
import type { Severity, Urgency, ManualAlertRequest } from '../../api/types'

interface ManualEntryFormProps {
  latitude: number | null
  longitude: number | null
  onSubmitted: () => void  // signal parent to clear map marker after success
}

const SEVERITIES: Severity[] = ['Extreme', 'Severe', 'Moderate', 'Minor']
const URGENCIES: Urgency[] = ['Immediate', 'Expected', 'Future']

interface FormState {
  headline: string
  severity: Severity
  urgency: Urgency
  description: string
  instruction: string
}

const initialForm: FormState = {
  headline: '',
  severity: 'Severe',
  urgency: 'Immediate',
  description: '',
  instruction: '',
}

export function ManualEntryForm({ latitude, longitude, onSubmitted }: ManualEntryFormProps) {
  const [form, setForm] = useState<FormState>(initialForm)
  const [submitting, setSubmitting] = useState(false)
  const [success, setSuccess] = useState<string | null>(null)
  const [error, setError] = useState<string | null>(null)
  // No useAlerts() here — Pipeline reconciliation is server-driven via the Socket.IO
  // `new_alert` event emitted by process_alert_logic. useAlerts is a per-instance
  // hook (no shared state container), so an addAlert() call from this form would
  // never reach the Pipeline component anyway.

  function bind<K extends keyof FormState>(key: K) {
    return {
      value: form[key],
      onChange: (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) =>
        setForm((f) => ({ ...f, [key]: e.target.value as FormState[K] })),
    }
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    if (submitting) return
    setSubmitting(true)
    setSuccess(null)
    setError(null)
    try {
      const payload: ManualAlertRequest = {
        headline: form.headline,
        severity: form.severity,
        urgency: form.urgency,
        description: form.description,
        instruction: form.instruction,
        // Empty string is OK — ghana_cap_app.py:357-366 _coord_or_default tolerates.
        latitude: latitude ?? '',
        longitude: longitude ?? '',
      }
      const res = await submitManualAlert(payload)
      setSuccess(`Submitted: ${res.identifier}. ${res.message}`)
      setForm(initialForm)
      onSubmitted()
      // The server emits `new_alert` after process_alert_logic completes;
      // <Pipeline>'s useAlerts subscription dedups by identifier and prepends.
      // No client-side optimistic update needed.
    } catch (e) {
      const msg = e instanceof ApiError ? `${e.status}: ${(e.payload as any)?.error ?? 'Submission failed'}` : String(e)
      setError(msg)
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-4" noValidate>
      <div>
        <label className="block text-sm font-medium mb-1" htmlFor="headline">Alert Headline</label>
        <input
          id="headline"
          type="text"
          required
          {...bind('headline')}
          placeholder="Enter a concise headline"
          className="w-full rounded-md bg-black/30 border border-white/15 p-2 text-sm focus:outline-none focus:ring-2 focus:ring-sky-400"
        />
      </div>

      <div className="grid grid-cols-2 gap-3">
        <div>
          <label className="block text-sm font-medium mb-1" htmlFor="severity">Severity</label>
          <select
            id="severity"
            {...bind('severity')}
            className="w-full rounded-md bg-black/30 border border-white/15 p-2 text-sm focus:outline-none focus:ring-2 focus:ring-sky-400"
          >
            {SEVERITIES.map((s) => <option key={s} value={s}>{s}</option>)}
          </select>
        </div>
        <div>
          <label className="block text-sm font-medium mb-1" htmlFor="urgency">Urgency</label>
          <select
            id="urgency"
            {...bind('urgency')}
            className="w-full rounded-md bg-black/30 border border-white/15 p-2 text-sm focus:outline-none focus:ring-2 focus:ring-sky-400"
          >
            {URGENCIES.map((u) => <option key={u} value={u}>{u}</option>)}
          </select>
        </div>
      </div>

      <div>
        <label className="block text-sm font-medium mb-1" htmlFor="description">Detailed Description</label>
        <textarea
          id="description"
          rows={5}
          required
          {...bind('description')}
          placeholder="Provide clear information about the event..."
          className="w-full rounded-md bg-black/30 border border-white/15 p-2 text-sm focus:outline-none focus:ring-2 focus:ring-sky-400"
        />
      </div>

      <div>
        <label className="block text-sm font-medium mb-1" htmlFor="instruction">Instructions for Public</label>
        <textarea
          id="instruction"
          rows={3}
          {...bind('instruction')}
          placeholder="What should the public do?"
          className="w-full rounded-md bg-black/30 border border-white/15 p-2 text-sm focus:outline-none focus:ring-2 focus:ring-sky-400"
        />
      </div>

      <div className="text-xs text-white/60">
        Coordinates: {latitude !== null && longitude !== null
          ? <>lat {latitude.toFixed(4)}, lon {longitude.toFixed(4)}</>
          : <span className="italic">Click the map to drop a marker (defaults to Accra if unset)</span>}
      </div>

      <button
        type="submit"
        disabled={submitting}
        className="w-full px-4 py-3 rounded-md bg-gradient-to-r from-sky-500 to-purple-500 text-white font-bold uppercase tracking-wide disabled:opacity-50"
      >
        {submitting ? 'PROCESSING…' : 'AUTHENTICATE & DISPATCH ALERT'}
      </button>

      {success && (
        <div role="status" className="p-3 rounded-md bg-emerald-500/10 border border-emerald-500/30 text-sm text-emerald-200">
          {success}
        </div>
      )}
      {error && (
        <div role="alert" className="p-3 rounded-md bg-red-500/10 border border-red-500/30 text-sm text-red-200">
          {error}
        </div>
      )}
    </form>
  )
}
```

What to AVOID and WHY:
- Do NOT use `react-leaflet` — PATTERNS.md "MapPanel.tsx" pins raw Leaflet so the Phase 5 Mapbox swap is local to one component. Adding `react-leaflet` doubles the surface that Phase 5 has to migrate.
- Do NOT add the GeoJSON upload button or hidden `<input type="file">` from the existing dashboard (`ghana_cap_dashboard.html:496-497`). It has no Phase 4 success criterion and no test in the existing suite — out of scope.
- Do NOT add `react-hook-form` — RESEARCH.md "Don't Hand-Roll" pins controlled inputs as the right fit for a 6-field form. Adding RHF inflates bundle size for no benefit.
- Do NOT bypass the ApiError flow with `try/catch (e: any)` — keep typed catch via `instanceof ApiError`, fall back to `String(e)` for non-API errors.
- Do NOT call `useAlerts()` inside `ManualEntryForm` — `useAlerts` is a per-component-instance hook (no shared state container), so an `addAlert()` call from this form would never reach the `<Pipeline>` instance. Pipeline reconciliation is server-driven via the Socket.IO `new_alert` event emitted by `process_alert_logic`; the event arrives within ~100ms with the full enriched alert document. Importing `useAlerts` here just to reference it is misleading dead code.

```typescript
import { useState } from 'react'
import { Zap } from 'lucide-react'
import { GlassCard } from '../ui/GlassCard'
import { testDispatch } from '../../api/system'
import { ApiError } from '../../api/client'
import type { Role } from '../../api/types'

interface TestDispatcherProps {
  userRole: Role | undefined
}

export function TestDispatcher({ userRole }: TestDispatcherProps) {
  const [submitting, setSubmitting] = useState(false)
  const [result, setResult] = useState<string | null>(null)
  const [error, setError] = useState<string | null>(null)

  if (userRole !== 'Admin') {
    return (
      <GlassCard variant="card">
        <h3 className="text-lg font-bold">Test Dispatcher</h3>
        <p className="text-sm text-white/60">Available to Admin role only.</p>
      </GlassCard>
    )
  }

  async function run() {
    if (submitting) return
    setSubmitting(true)
    setResult(null)
    setError(null)
    try {
      const res = await testDispatch()
      setResult(`Mock alert dispatched: ${res.identifier}. Check the Pipeline tab.`)
    } catch (e) {
      const msg = e instanceof ApiError ? `${e.status}: ${(e.payload as any)?.error ?? 'Test dispatch failed'}` : String(e)
      setError(msg)
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <GlassCard variant="card" className="space-y-3">
      <h3 className="text-lg font-bold">Test Dispatcher</h3>
      <p className="text-sm text-white/70">
        Inject a synthetic GMeT payload through the full ingest → enrich → dispatch pipeline. Useful for verifying end-to-end without waiting for a real weather event.
      </p>
      <button
        type="button"
        onClick={run}
        disabled={submitting}
        className="inline-flex items-center gap-2 px-4 py-2 rounded-md bg-sky-500/20 hover:bg-sky-500/30 border border-sky-500/40 text-sm font-bold uppercase tracking-wide disabled:opacity-50"
      >
        <Zap className="h-4 w-4" />
        {submitting ? 'Dispatching…' : 'Run Test Dispatch'}
      </button>
      {result && (
        <div role="status" className="text-sm text-emerald-200 p-2 rounded-md bg-emerald-500/10 border border-emerald-500/30">
          {result}
        </div>
      )}
      {error && (
        <div role="alert" className="text-sm text-red-200 p-2 rounded-md bg-red-500/10 border border-red-500/30">
          {error}
        </div>
      )}
    </GlassCard>
  )
}
```
